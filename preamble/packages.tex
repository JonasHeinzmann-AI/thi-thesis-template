% Zotero titles routinely contain raw Unicode subscripts/superscripts (e.g. a
% literal "T₂"). Better BibTeX passes them through verbatim, and pdfLaTeX errors
% on them. Map the ranges to text sub/superscripts so the bibliography compiles
% whatever the library contains -- references.bib must not be hand-edited.
\usepackage{newunicodechar}
\newunicodechar{₀}{\textsubscript{0}}
\newunicodechar{₁}{\textsubscript{1}}
\newunicodechar{₂}{\textsubscript{2}}
\newunicodechar{₃}{\textsubscript{3}}
\newunicodechar{₄}{\textsubscript{4}}
\newunicodechar{ρ}{\ensuremath{\rho}}
\newunicodechar{×}{\ensuremath{\times}}
\newunicodechar{≤}{\ensuremath{\leq}}
\newunicodechar{≥}{\ensuremath{\geq}}
\newunicodechar{−}{\ensuremath{-}}
