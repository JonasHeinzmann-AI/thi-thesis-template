% Margins are set explicitly by `geometry` in preamble/packages.tex
% (left 30mm / right 20mm, the THI convention). KOMA's DIV/BCOR are therefore
% deliberately NOT used -- the two mechanisms conflict.
% `bibliography=totoc` is likewise omitted: biblatex handles the ToC entry via
% \printbibliography[heading=bibintoc]. Using both duplicates it.
% Body text is 12pt -- the usual size for a German thesis. If you prefer a
% tighter page, 11pt is KOMA's own default; heading sizes in preamble/style.tex
% scale with it automatically and need no adjustment.
\documentclass[
    a4paper,
    12pt,               % -> 11pt for a tighter page
    oneside,            % -> twoside for double-sided printing
    parskip=half,
    listof=totoc,
    numbers=noenddot,
    english
]{scrbook}

\input{metadata}
\input{preamble/packages}
\input{preamble/style}

\begin{document}

\tableofcontents
\listoffigures
\listoftables
\printacronyms[title={List of Abbreviations}]
\input{frontmatter/symbols}
